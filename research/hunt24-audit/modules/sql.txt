    rmf."source",
    mmhdd.x_match_tag,
    mmhdd.audit_status,
    mmhdd.kine_chi2,

	mmhdd.rv_chi2,
	mmhdd.cmd_chi2,

    rmf.de_icrs,
    rmf.pmra,
    rmf.pmde,
    rmf.e_pmra,
    rmf.e_pmde,
    rmf.gmag,
    rmf.color,
    rmf.e_color,
    rmf.corrected_color_excess,

INNER JOIN astrodb_internal.main.master_m45_hunt_5d_dbscan mmhdd
    ON rmf."source" = mmhdd.id
	AND mmhdd.x_match_tag IS NOT NULL;

FROM astrodb_internal.main.raw_m45_field rmf 
WHERE rmf.Plx > 4
	AND rmf.pmRA BETWEEN 0 AND 40
	AND rmf.pmDE BETWEEN -60 AND -20
	AND rmf.RUWE < 1.4

SELECT 
    ROUND(FLOOR(mmhdhd.prob / 0.1) * 0.1, 1) AS bin_start,
    CONCAT(ROUND(FLOOR(mmhdhd.prob / 0.1) * 0.1, 1), ' - ', ROUND(FLOOR(mmhdhd.prob / 0.1) * 0.1 + 0.09, 2)) AS bin_range,
    COUNT(*) AS frequency
FROM astrodb_internal.main.master_m45_hunt_5d_h_dbscan mmhdhd 
WHERE mmhdhd.prob IS NOT NULL
GROUP BY FLOOR(mmhdhd.prob / 0.1)
ORDER BY bin_start;
